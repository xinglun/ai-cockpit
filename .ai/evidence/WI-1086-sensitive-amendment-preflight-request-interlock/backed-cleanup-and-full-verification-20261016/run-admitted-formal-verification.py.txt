def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def free():
    return {p: os.statvfs(p).f_bavail * os.statvfs(p).f_frsize for p in ['/workspace', '/tmp']}

def record(name, value):
    with (LOG / name).open('x') as f:
        json.dump(value, f, ensure_ascii=False, indent=2)
        f.write('\n')

def descendants():
    seen, queue = {}, [os.getpid()]
    while queue:
        parent = queue.pop()
        try:
            children = {int(x) for task in Path(f'/proc/{parent}/task').iterdir()
                        for x in (task / 'children').read_text().split()}
        except OSError:
            continue
        for pid in children:
            if pid in seen or pid == os.getpid():
                continue
            try:
                raw = Path(f'/proc/{pid}/stat').read_text()
                fields = raw[raw.rfind(')') + 2:].split()
                seen[pid] = {'pid': pid, 'startTimeTicks': fields[19], 'state': fields[0],
                             'comm': raw[raw.find('(') + 1:raw.rfind(')')]}
                queue.append(pid)
            except OSError:
                pass
    return seen

reaped = []
def reap(protected=None):
    try:
        children = {int(x) for task in Path(f'/proc/{os.getpid()}/task').iterdir()
                    for x in (task / 'children').read_text().split()}
    except OSError:
        return
    for pid in children:
        if pid == protected:
            continue
        try:
            child, status = os.waitpid(pid, os.WNOHANG)
            if child:
                reaped.append({'pid': child, 'waitStatus': status})
        except ChildProcessError:
            pass

def terminate_owned(sig):
    targets = descendants()
    for pid, item in reversed(list(targets.items())):
        try:
            raw = Path(f'/proc/{pid}/stat').read_text()
            fields = raw[raw.rfind(')') + 2:].split()
            if fields[19] == item['startTimeTicks'] and fields[0] != 'Z':
                os.kill(pid, sig)
        except ProcessLookupError:
            pass
        except FileNotFoundError:
            pass
    return list(targets.values())

def query(label, args):
    command = [str(RUNTIME), *args]
    limit = min(120, max(1, 880 - (time.monotonic() - clock)))
    p = subprocess.run(command, cwd=REPO, env=ENV, capture_output=True, timeout=limit)
    (LOG / (label + '.stdout')).write_bytes(p.stdout)
    (LOG / (label + '.stderr')).write_bytes(p.stderr)
    record(label + '.command.json', {'argv': command, 'exitCode': p.returncode})
    reap()
    assert p.returncode == 0, label + ': query failed; no verification started'
    return json.loads(p.stdout)

def status(label):
    value = query(label, ['work-item', 'status', '--repo', str(REPO), '--id', WI, '--json'])
    assert value['humanDecisionRequired'] is False
    assert value['actionExplanation']['admissionState'] == 'allowed'
    assert 'run_verification' in value['safeActions']
    assert value['sourceDigests']['contract'] == CONTRACT
    assert value['sourceDigests']['repositorySnapshot'] == SNAPSHOT
    return value

def monitor():

    while not stop_monitor.is_set():

        for name, value in values.items():

        reason = None
        if min(values.values()) <= FLOOR:
            reason = 'disk_floor_512MiB'
        elif sum(p.stat().st_size for p in LOG.iterdir() if p.is_file()) >= 64 * 1024 * 1024:
            reason = 'new_log_budget_64MiB'
        elif time.monotonic() - clock >= 885:
            reason = 'outer_deadline_reserving_15s_for_cleanup'
        if reason and meta['stopReason'] is None:
            meta['stopReason'] = reason
            meta['cancelRequestedAtElapsedSeconds'] = time.monotonic() - clock
            meta['terminatedOwnedProcesses'] = terminate_owned(signal.SIGTERM)
        if meta['stopReason'] and time.monotonic() - clock - meta['cancelRequestedAtElapsedSeconds'] >= 5:
            meta['killedOwnedProcesses'] = terminate_owned(signal.SIGKILL)
        reap(process.pid if process else None)
        if time.monotonic() - last_progress >= 45:
            last_progress = time.monotonic()
            owned = descendants()
            cargo_args = []
            for pid, item in owned.items():
                if item['comm'] == 'cargo':
                    try:
                        args = Path(f'/proc/{pid}/cmdline').read_bytes().split(b'\0')
                        cargo_args.append([a.decode(errors='replace') for a in args[:7]])
                    except OSError:
                        pass
            progress = {'elapsedSeconds': round(time.monotonic() - clock, 1),
                        'freeBytes': values, 'ownedProcessCount': len(owned),
                        'activeCargoArguments': cargo_args, 'stopReason': meta['stopReason']}
            meta['progressObservations'].append(progress)
            print(json.dumps({'verificationProgress': progress}), flush=True)
