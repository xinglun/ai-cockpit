def save(name, value):
    p = LOG / name
    with p.open('x') as f:
        json.dump(value, f, ensure_ascii=False, indent=2)
        f.write('\n')
        f.flush()
        os.fsync(f.fileno())

def disks():
    return {name: {'device': Path(name).stat().st_dev,
                   'availableBytes': os.statvfs(name).f_bavail * os.statvfs(name).f_frsize,
                   'availableInodes': os.statvfs(name).f_favail} for name in ['/workspace', '/tmp']}

def digest_file(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def parent_fd(path):
    assert path.is_absolute() and '..' not in path.parts
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY)
    try:
        for part in path.parent.parts[1:]:
            nxt = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = nxt
        return fd
    except BaseException:
        os.close(fd)
        raise

def matches(s, row):
    return stat.S_ISREG(s.st_mode) and (
        s.st_dev, s.st_ino, stat.S_IMODE(s.st_mode), s.st_uid, s.st_gid,
        s.st_size, s.st_mtime_ns, s.st_ctime_ns, s.st_nlink
    ) == (row['device'], row['inode'], int(row['mode'], 8), row['uid'], row['gid'],
          row['sizeBytes'], row['mtimeNs'], row['ctimeNs'], row['linkCount'])

def open_checked(row):
    path = Path(row['path'])
    relative = path.relative_to(ROOT)
    assert str(relative) == row['relativePath'] and '..' not in relative.parts
    directory = parent_fd(path)
    try:
        fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=directory)
        assert matches(os.fstat(fd), row), str(path)
        h = hashlib.sha256()
        for chunk in iter(lambda: os.read(fd, 1024 * 1024), b''):
            h.update(chunk)
        assert h.hexdigest() == row['sha256'] and matches(os.fstat(fd), row), str(path)
        os.lseek(fd, 0, os.SEEK_SET)
        return directory, fd
    except BaseException:
        try:
            os.close(fd)
        except (UnboundLocalError, OSError):
            pass
        os.close(directory)
        raise

def observe():
    processes, unknowns, references = [], [], []
    for p in Path('/proc').iterdir():
        if not p.name.isdigit():
            continue
        try:
            raw = (p / 'stat').read_text()
            fields = raw[raw.rfind(')') + 2:].split()
            comm = raw[raw.find('(') + 1:raw.rfind(')')]
            if fields[0] == 'Z':
                continue
            processes.append({'pid': int(p.name), 'comm': comm, 'state': fields[0]})
        except OSError as e:
            if e.errno != 2:
                unknowns.append({'pid': int(p.name), 'comm': '', 'kind': 'stat/status', 'errno': e.errno})
            continue
        for kind in ['exe', 'cwd', 'root', 'maps', 'fd']:
            try:
                if kind == 'maps':
                    for line in (p / kind).read_text().splitlines():
                        if str(ROOT) in line:
                            references.append({'pid': int(p.name), 'kind': kind, 'reference': line})
                elif kind == 'fd':
                    for f in (p / kind).iterdir():
                        try:
                            target = os.readlink(f)
                            if str(ROOT) in target and int(p.name) != os.getpid():
                                references.append({'pid': int(p.name), 'kind': kind, 'reference': target})
                        except OSError as e:
                            if e.errno != 2:
                                unknowns.append({'pid': int(p.name), 'comm': comm, 'kind': 'fd/readlink', 'errno': e.errno})
                else:
                    target = os.readlink(p / kind)
                    if str(ROOT) in target:
                        references.append({'pid': int(p.name), 'kind': kind, 'reference': target})
            except OSError as e:
                if e.errno != 2:
                    unknowns.append({'pid': int(p.name), 'comm': comm, 'kind': kind, 'errno': e.errno})
    base = json.loads((INVENTORY / 'occupancy-observation.json').read_text())['unknowns']
    def key(item):
        return (item['pid'], item.get('comm', ''), 'fd' if item['kind'].startswith('fd') else item['kind'], item['errno'])
    approved = {key(x) for x in base}
    new = [x for x in unknowns if key(x) not in approved]
    return {'observedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
            'liveProcesses': processes, 'targetReferencesObserved': references,
            'unreadableReferenceUnknowns': unknowns, 'newUndisclosedUnknowns': new,
            'riskAcceptedByRay': True, 'occupancyVerified': False,
            'acceptedRiskScope': 'only previously disclosed unreadable-process reference uncertainty'}

def assert_no_activity(observation):
    assert not observation['targetReferencesObserved'], 'actual target reference observed'
    assert not observation['newUndisclosedUnknowns'], 'new undisclosed process observation uncertainty'
    assert not [x for x in observation['liveProcesses'] if x['comm'] in ['cargo', 'rustc']], 'active build observed'
